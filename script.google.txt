// ============================================
// Ceedling Docker Version Checker
//
// Docker Hub:
//   safdariali/ceedling
//
// Tag format:
//   CEEDLING-ruby-RUBY-alpine-ALPINE
//
// Example:
//   1.1.4-ruby-3.4.10-alpine-3.24
//
// Policy:
//
//   Ceedling -> automatic update
//   Ruby     -> pinned
//   Alpine   -> pinned
// ============================================


// ============================================
// Configuration
// ============================================

const CEEDLING_DOCKER_USERNAME =
  'safdariali';

const CEEDLING_DOCKER_REPOSITORY =
  'ceedling';


// --------------------------------------------
// Pinned build environment
// --------------------------------------------

const PINNED_RUBY_VERSION =
  '3.4.10';

const PINNED_ALPINE_VERSION =
  '3.24';


// --------------------------------------------
// GitHub repository
// --------------------------------------------

const GITHUB_REPOSITORY =
  'SafdariAli/ceedling-docker';

const GITHUB_WORKFLOW =
  'docker.yml';

const GITHUB_BRANCH =
  'main';


// ============================================
// Function 1:
// Compare semantic versions
// ============================================

function compareVersions(a, b) {

  const partsA =
    a.split('.').map(Number);

  const partsB =
    b.split('.').map(Number);


  const length =
    Math.max(
      partsA.length,
      partsB.length
    );


  for (
    let i = 0;
    i < length;
    i++
  ) {

    const valueA =
      partsA[i] || 0;

    const valueB =
      partsB[i] || 0;


    if (
      valueA !== valueB
    ) {

      return valueA - valueB;
    }
  }


  return 0;
}


// ============================================
// Function 2:
// Get latest Ceedling version
// from RubyGems
// ============================================

function getCeedlingVersion() {

  try {

    Logger.log(
      'Fetching the latest Ceedling version from RubyGems...'
    );


    const url =
      'https://rubygems.org/api/v1/gems/ceedling.json';


    const response =
      UrlFetchApp.fetch(
        url,
        {
          muteHttpExceptions:
            true,

          headers: {
            'Accept':
              'application/json'
          }
        }
      );


    const statusCode =
      response.getResponseCode();


    if (
      statusCode !== 200
    ) {

      throw new Error(
        'RubyGems API returned HTTP ' +
        statusCode
      );
    }


    const data =
      JSON.parse(
        response.getContentText()
      );


    if (
      !data.version
    ) {

      throw new Error(
        'Ceedling version not found in RubyGems response'
      );
    }


    Logger.log(
      '✅ Latest Ceedling version: ' +
      data.version
    );


    return data.version;


  } catch (error) {

    Logger.log(
      '❌ Error fetching Ceedling version: ' +
      error.toString()
    );


    return null;
  }
}


// ============================================
// Function 3:
// Get current versions from Docker Hub
//
// Repository:
//
//   safdariali/ceedling
//
// Expected tag:
//
//   1.1.4-ruby-3.4.10-alpine-3.24
//
// IMPORTANT:
//
// Docker Hub API errors are NOT treated as
// an empty repository.
// ============================================

function getCurrentVersions() {

  try {

    Logger.log(
      'Fetching current Ceedling versions from Docker Hub...'
    );


    const pageSize =
      100;


    let url =
      `https://hub.docker.com/v2/namespaces/${CEEDLING_DOCKER_USERNAME}` +
      `/repositories/${CEEDLING_DOCKER_REPOSITORY}` +
      `/tags?page_size=${pageSize}`;


    let currentCeedling =
      null;

    let currentRuby =
      null;

    let currentAlpine =
      null;


    let pageCount =
      0;

    let totalTags =
      0;


    while (
      url
    ) {

      pageCount++;


      Logger.log(
        'Fetching Docker Hub tags page ' +
        pageCount +
        '...'
      );


      const response =
        UrlFetchApp.fetch(
          url,
          {
            muteHttpExceptions:
              true,

            headers: {
              'Accept':
                'application/json'
            }
          }
        );


      const statusCode =
        response.getResponseCode();


      // ----------------------------------------
      // API ERROR
      // ----------------------------------------

      if (
        statusCode !== 200
      ) {

        throw new Error(
          'Docker Hub API returned HTTP ' +
          statusCode +
          ': ' +
          response.getContentText()
        );
      }


      const data =
        JSON.parse(
          response.getContentText()
        );


      // ----------------------------------------
      // Count tags
      // ----------------------------------------

      if (
        data.results
      ) {

        totalTags +=
          data.results.length;
      }


      // ----------------------------------------
      // No results
      // ----------------------------------------

      if (
        !data.results ||
        data.results.length === 0
      ) {

        break;
      }


      // ----------------------------------------
      // Inspect tags
      // ----------------------------------------

      for (
        const result of data.results
      ) {

        const tag =
          result.name;


        // Expected:
        //
        // 1.1.4-ruby-3.4.10-alpine-3.24
        //
        // Also supports:
        //
        // 1.1.4-ruby-3.4.10-alpine-3.24.1

        const match =
          tag.match(
            /^(\d+\.\d+\.\d+)-ruby-(\d+\.\d+\.\d+)-alpine-(\d+\.\d+(?:\.\d+)?)$/
          );


        if (
          !match
        ) {

          continue;
        }


        const ceedlingVersion =
          match[1];

        const rubyVersion =
          match[2];

        const alpineVersion =
          match[3];


        // --------------------------------------
        // Find newest Ceedling version
        // --------------------------------------

        if (
          currentCeedling === null
        ) {

          currentCeedling =
            ceedlingVersion;

          currentRuby =
            rubyVersion;

          currentAlpine =
            alpineVersion;

          continue;
        }


        if (
          compareVersions(
            ceedlingVersion,
            currentCeedling
          ) > 0
        ) {

          currentCeedling =
            ceedlingVersion;

          currentRuby =
            rubyVersion;

          currentAlpine =
            alpineVersion;
        }
      }


      // ----------------------------------------
      // Pagination
      // ----------------------------------------

      url =
        data.next || null;
    }


    // ----------------------------------------
    // Docker Hub reachable
    // ----------------------------------------

    Logger.log(
      '✅ Docker Hub API reachable'
    );


    Logger.log(
      '📦 Total tags inspected: ' +
      totalTags
    );


    // ----------------------------------------
    // Repository exists but no version tags
    // ----------------------------------------

    if (
      currentCeedling === null
    ) {

      if (
        totalTags === 0
      ) {

        Logger.log(
          'ℹ️ Docker repository exists but contains no tags'
        );

      } else {

        Logger.log(
          '⚠️ Docker repository contains tags, but no valid Ceedling version tags were found'
        );
      }


      return {

        status:
          'empty',

        ceedling:
          null,

        ruby:
          null,

        alpine:
          null,

        totalTags:
          totalTags
      };
    }


    // ----------------------------------------
    // Valid image found
    // ----------------------------------------

    Logger.log(
      `✅ Current versions - Ceedling: ${currentCeedling}, Ruby: ${currentRuby}, Alpine: ${currentAlpine}`
    );


    return {

      status:
        'success',

      ceedling:
        currentCeedling,

      ruby:
        currentRuby,

      alpine:
        currentAlpine,

      totalTags:
        totalTags
    };


  } catch (error) {

    // ----------------------------------------
    // IMPORTANT:
    //
    // This is an API/network error.
    //
    // It is NOT an empty repository.
    // ----------------------------------------

    Logger.log(
      '❌ Docker Hub API error: ' +
      error.toString()
    );


    return {

      status:
        'error',

      ceedling:
        null,

      ruby:
        null,

      alpine:
        null,

      error:
        error.toString()
    };
  }
}


// ============================================
// Function 4:
// Main version check and build
// ============================================

function checkAndBuild() {

  try {

    Logger.log(
      '========================================'
    );

    Logger.log(
      '🔄 Starting comprehensive version check'
    );

    Logger.log(
      '========================================'
    );


    // ----------------------------------------
    // 1. Get latest Ceedling
    // ----------------------------------------

    const latestCeedling =
      getCeedlingVersion();


    if (
      !latestCeedling
    ) {

      throw new Error(
        '❌ Failed to fetch latest Ceedling version'
      );
    }


    // ----------------------------------------
    // 2. Pinned Ruby + Alpine
    // ----------------------------------------

    const latestRuby =
      PINNED_RUBY_VERSION;

    const latestAlpine =
      PINNED_ALPINE_VERSION;


    Logger.log(
      '📌 Build environment is pinned:'
    );

    Logger.log(
      '   Ruby: ' +
      latestRuby
    );

    Logger.log(
      '   Alpine: ' +
      latestAlpine
    );


    Logger.log(
      `📦 Target versions - Ceedling: ${latestCeedling}, Ruby: ${latestRuby}, Alpine: ${latestAlpine}`
    );


    // ----------------------------------------
    // 3. Get current Docker image
    // ----------------------------------------

    const currentVersions =
      getCurrentVersions();


    // ========================================
    // SAFETY CHECK
    //
    // Docker Hub API error:
    //
    // DO NOT BUILD
    // DO NOT MODIFY latest
    // ========================================

    if (
      currentVersions.status ===
      'error'
    ) {

      throw new Error(
        '❌ Docker Hub could not be checked. ' +
        'Build aborted for safety.\n' +
        currentVersions.error
      );
    }


    // ----------------------------------------
    // 4. Initial build
    // ----------------------------------------

    if (
      currentVersions.status ===
      'empty'
    ) {

      Logger.log(
        '🚀 Docker repository has no versioned image. Building first version...'
      );


      const result =
        triggerGitHubBuild(
          latestCeedling,
          latestRuby,
          latestAlpine
        );


      if (
        !result
      ) {

        throw new Error(
          '❌ Failed to trigger initial build'
        );
      }


      Logger.log(
        '✅ Initial build triggered successfully'
      );


      return {

        success:
          true,

        ceedling:
          latestCeedling,

        ruby:
          latestRuby,

        alpine:
          latestAlpine,

        message:
          `Initial build for Ceedling ${latestCeedling} started`
      };
    }


    // ----------------------------------------
    // 5. Compare versions
    // ----------------------------------------

    const ceedlingChanged =
      currentVersions.ceedling !==
      latestCeedling;


    const rubyChanged =
      currentVersions.ruby !==
      latestRuby;


    const alpineChanged =
      currentVersions.alpine !==
      latestAlpine;


    Logger.log(
      `📊 Comparison:
      Ceedling: ${currentVersions.ceedling} → ${latestCeedling} ${ceedlingChanged ? '🔴 CHANGED' : '✅ Same'}
      Ruby:     ${currentVersions.ruby} → ${latestRuby} ${rubyChanged ? '🔴 CHANGED' : '✅ Same'}
      Alpine:   ${currentVersions.alpine} → ${latestAlpine} ${alpineChanged ? '🔴 CHANGED' : '✅ Same'}`
    );


    // ----------------------------------------
    // 6. Build if anything changed
    // ----------------------------------------

    if (
      ceedlingChanged ||
      rubyChanged ||
      alpineChanged
    ) {

      Logger.log(
        '========================================'
      );

      Logger.log(
        '🚨 New version/environment detected!'
      );

      Logger.log(
        '🚀 Building new Ceedling Docker image...'
      );


      if (
        ceedlingChanged
      ) {

        Logger.log(
          `  Ceedling: ${currentVersions.ceedling} → ${latestCeedling}`
        );
      }


      if (
        rubyChanged
      ) {

        Logger.log(
          `  Ruby: ${currentVersions.ruby} → ${latestRuby}`
        );
      }


      if (
        alpineChanged
      ) {

        Logger.log(
          `  Alpine: ${currentVersions.alpine} → ${latestAlpine}`
        );
      }


      Logger.log(
        '========================================'
      );


      const result =
        triggerGitHubBuild(
          latestCeedling,
          latestRuby,
          latestAlpine
        );


      if (
        !result
      ) {

        throw new Error(
          '❌ Failed to trigger GitHub Action'
        );
      }


      Logger.log(
        '✅ GitHub Action triggered successfully'
      );


      return {

        success:
          true,

        ceedling:
          latestCeedling,

        ruby:
          latestRuby,

        alpine:
          latestAlpine,

        changes: {

          ceedling:
            ceedlingChanged,

          ruby:
            rubyChanged,

          alpine:
            alpineChanged
        },

        message:
          `Build started for Ceedling ${latestCeedling}, Ruby ${latestRuby}, Alpine ${latestAlpine}`
      };


    } else {

      // ----------------------------------------
      // Everything is up to date
      // ----------------------------------------

      Logger.log(
        '========================================'
      );

      Logger.log(
        '✅ All versions are up to date!'
      );

      Logger.log(
        `  Ceedling: ${currentVersions.ceedling}`
      );

      Logger.log(
        `  Ruby:     ${currentVersions.ruby}`
      );

      Logger.log(
        `  Alpine:   ${currentVersions.alpine}`
      );

      Logger.log(
        '========================================'
      );


      return {

        success:
          true,

        ceedling:
          currentVersions.ceedling,

        ruby:
          currentVersions.ruby,

        alpine:
          currentVersions.alpine,

        message:
          'All versions are up to date'
      };
    }


  } catch (error) {

    Logger.log(
      '❌ Error: ' +
      error.toString()
    );


    sendErrorEmail(
      error.toString()
    );


    return {

      success:
        false,

      error:
        error.toString()
    };
  }
}


// ============================================
// Function 5:
// Trigger GitHub Actions
// ============================================

function triggerGitHubBuild(
  ceedlingVersion,
  rubyVersion,
  alpineVersion
) {

  try {

    Logger.log(
      '🚀 Triggering GitHub Action...'
    );


    Logger.log(
      `  Ceedling: ${ceedlingVersion}`
    );

    Logger.log(
      `  Ruby:     ${rubyVersion}`
    );

    Logger.log(
      `  Alpine:   ${alpineVersion}`
    );


    const token =
      PropertiesService
        .getScriptProperties()
        .getProperty(
          'GITHUB_TOKEN'
        );


    if (
      !token
    ) {

      throw new Error(
        '❌ GITHUB_TOKEN is not set'
      );
    }


    const url =
      `https://api.github.com/repos/${GITHUB_REPOSITORY}` +
      `/actions/workflows/${GITHUB_WORKFLOW}/dispatches`;


    const payload = {

      ref:
        GITHUB_BRANCH,

      inputs: {

        ceedling_version:
          ceedlingVersion,

        ruby_version:
          rubyVersion,

        alpine_version:
          alpineVersion
      }
    };


    const options = {

      method:
        'POST',

      headers: {

        'Authorization':
          `token ${token}`,

        'Accept':
          'application/vnd.github.v3+json',

        'Content-Type':
          'application/json'
      },

      payload:
        JSON.stringify(payload),

      muteHttpExceptions:
        true
    };


    const response =
      UrlFetchApp.fetch(
        url,
        options
      );


    const statusCode =
      response.getResponseCode();


    if (
      statusCode === 204
    ) {

      Logger.log(
        '✅ GitHub Action triggered successfully'
      );


      return true;
    }


    throw new Error(
      `GitHub API error: Status code ${statusCode}, Response: ${response.getContentText()}`
    );


  } catch (error) {

    Logger.log(
      '❌ Error triggering build: ' +
      error.toString()
    );


    return false;
  }
}


// ============================================
// Function 6:
// Send error email
// ============================================

function sendErrorEmail(
  errorMessage
) {

  try {

    const email =
      PropertiesService
        .getScriptProperties()
        .getProperty(
          'NOTIFICATION_EMAIL'
        );


    if (
      !email
    ) {

      Logger.log(
        '⚠️ NOTIFICATION_EMAIL is not set in Script Properties'
      );

      return;
    }


    const subject =
      '❌ Ceedling Docker Build Error';


    const body = `

An error occurred during the Ceedling version check and build process.

Error:
${errorMessage}

Time:
${new Date().toISOString()}

Please check Google Apps Script logs.

    `;


    MailApp.sendEmail({

      to:
        email,

      subject:
        subject,

      body:
        body
    });


    Logger.log(
      '✅ Error email sent'
    );


  } catch (error) {

    Logger.log(
      '❌ Error sending email: ' +
      error.toString()
    );
  }
}


// ============================================
// Function 7:
// Setup daily automatic check
// ============================================

function setupDailyCheck() {

  const triggers =
    ScriptApp.getProjectTriggers();


  triggers.forEach(
    trigger => {

      if (
        trigger.getHandlerFunction() ===
        'checkAndBuild'
      ) {

        ScriptApp.deleteTrigger(
          trigger
        );
      }
    }
  );


  ScriptApp
    .newTrigger(
      'checkAndBuild'
    )
    .timeBased()
    .everyDays(1)
    .atHour(9)
    .create();


  Logger.log(
    '✅ Daily trigger set for 9 AM'
  );
}


// ============================================
// Function 8:
// Manual run
// ============================================

function runManualCheck() {

  Logger.log(
    '========================================'
  );

  Logger.log(
    '🔄 Manual version check started'
  );

  Logger.log(
    '========================================'
  );


  const result =
    checkAndBuild();


  Logger.log(
    '========================================'
  );

  Logger.log(
    '📊 Result:'
  );

  Logger.log(
    JSON.stringify(
      result,
      null,
      2
    )
  );

  Logger.log(
    '========================================'
  );


  return result;
}
